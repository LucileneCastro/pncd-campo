const CACHE_APP = "pncd-app-v1";
const CACHE_MAPA = "pncd-tiles-v1";

const ARQUIVOS_BASE = [
  "./",
  "./index.html",
  "./db.js",
  "./map.js",
  "./app.js",
  "./manifest.json",
  "https://unpkg.com/dexie@4.0.8/dist/dexie.min.js",
  "https://unpkg.com/leaflet@1.9.4/dist/leaflet.css",
  "https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"
];

self.addEventListener("install", (event) => {
  event.waitUntil(
    caches.open(CACHE_APP).then((cache) => cache.addAll(ARQUIVOS_BASE))
  );
});

self.addEventListener("fetch", (event) => {
  const url = event.request.url;

  // Cache dinâmico para blocos de mapa (tiles)
  if (url.includes("tile.openstreetmap.org")) {
    event.respondWith(
      caches.open(CACHE_MAPA).then(async (cache) => {
        const cached = await cache.match(event.request);
        if (cached) return cached;
        try {
          const response = await fetch(event.request);
          cache.put(event.request, response.clone());
          return response;
        } catch {
          return new Response("", { status: 408 });
        }
      })
    );
    return;
  }

  // Cache-first para arquivos do aplicativo
  event.respondWith(
    caches.match(event.request).then((resp) => resp || fetch(event.request))
  );
});
